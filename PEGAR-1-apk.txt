name: Construir APK
on:
  workflow_dispatch:
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - name: Copiar la web a www/
        run: |
          mkdir -p www
          cp index.html styles.css app.js ui.js data.js sw.js manifest.webmanifest icon.svg news.json rules.json www/
          sed -i "s#const NEWS_URL = 'news.json';#const NEWS_URL = 'https://${{ github.repository_owner }}.github.io/${{ github.event.repository.name }}/news.json';#" www/ui.js
          sed -i "s#const RULES_URL = 'rules.json';#const RULES_URL = 'https://${{ github.repository_owner }}.github.io/${{ github.event.repository.name }}/rules.json';#" www/data.js
          grep -n "_URL = " www/ui.js www/data.js
      - name: Crear proyecto Android
        run: |
          npm install
          npx cap add android
          npx cap sync android
          sed -i 's#</manifest>#    <uses-permission android:name="android.permission.ACCESS_FINE_LOCATION" />\n    <uses-permission android:name="android.permission.ACCESS_COARSE_LOCATION" />\n</manifest>#' android/app/src/main/AndroidManifest.xml
      - name: Compilar
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon
      - uses: actions/upload-artifact@v4
        with:
          name: dron-espana-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
